"UP": insights.append({ "title": f"{metric.name.title()} is growing", "detail": f"{metric.name} increased by {pct:+.2f}% from Q1 to Q2, signaling strong momentum.", "tone": "positive", }) elif metric.trend() == "DOWN": insights.append({ "title": f"{metric.name.title()} needs attention", "detail": f"{metric.name} declined by {pct:+.2f}% from Q1 to Q2 and should be monitored closely.", "tone": "warning", }) else: insights.append({ "title": f"{metric.name.title()} is stable", "detail": f"{metric.name} remained flat, showing stable behavior across the two quarters.", "tone": "neutral", }) return insights[:5] def load_analysis(csv_path: str = "data/metrics.csv"): path = Path(csv_path) if not path.exists(): raise FileNotFoundError(f"Dataset not found: {csv_path}") rows =
